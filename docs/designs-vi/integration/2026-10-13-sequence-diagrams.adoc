order ->> bus: [EventBridge/SQS async] publish OrderPaid.v1
bus ->> int: [EventBridge/SQS async] deliver OrderPaid.v1
int -> db: [SQL] Khử trùng lặp inbox + tạo tác vụ hóa đơn
db --> int: REQUESTED / tác vụ đã tồn tại
alt Trùng lặp hoặc đã phát hành trước đó
  int -> db: [SQL] Đọc tác vụ hiện tại
  db --> int: Trạng thái hiện có
else Tác vụ mới
  int ->> bus: [EventBridge/SQS async] InvoiceOperationRequested.v1
  int -> provider: [HTTP/REST] Phát hành hóa đơn kỹ thuật + idempotency key
  alt Provider chấp nhận
    provider --> int: 200 provider ref
    int -> db: [SQL] Commit ISSUED + lượt thử provider + outbox
    int ->> bus: [EventBridge/SQS async] InvoiceIssued.v1
  else Timeout/lỗi có thể thử lại (retryable failure)
    provider --> int: Timeout/429/5xx
    int -> db: [SQL] Commit FAILED_RETRYABLE + lượt thử tiếp theo
  else Lỗi cuối cùng (terminal failure)
    provider --> int: 4xx/payload không hỗ trợ
    int -> db: [SQL] Commit FAILED_TERMINAL + mã lỗi an toàn
    int ->> bus: [EventBridge/SQS async] InvoiceFailed.v1
  end
end

    zalo --> int: 202 provider message ref
    int -> db: [SQL] Commit SENT + trạng thái outbox

    zalo --> int: Kết quả provider
    int -> db: [SQL] Commit trạng thái đã chuẩn hóa
  end
end

  int -> db: [SQL] Duy trì trạng thái thử lại/đối soát

else Provider báo lỗi dứt khoát
  provider --> int: FAILED
  int -> db: [SQL] Commit trạng thái chuẩn hóa lỗi cuối cùng/có thể thử lại
end

@enduml
